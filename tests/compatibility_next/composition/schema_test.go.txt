package schema_test

import (
 "encoding/json"
 "os"
 "path/filepath"
 "testing"
 "github.com/santhosh-tekuri/jsonschema/v5"
)

func TestFinalV040CompositionAudit(t *testing.T) {
 directory:=os.Getenv("DIRCUE_COMPOSITION_REPORTS")
 if directory=="" {t.Skip("set DIRCUE_COMPOSITION_REPORTS to an absolute composition output directory")}
 compiled,err:=jsonschema.Compile("profile.schema.json");if err!=nil{t.Fatal(err)}
 paths,err:=filepath.Glob(filepath.Join(directory,"composition-*.json"));if err!=nil{t.Fatal(err)}
 count:=0
 for _,path:=range paths {
  content,err:=os.ReadFile(path);if err!=nil{t.Fatal(err)}
  var report map[string]any;if err=json.Unmarshal(content,&report);err!=nil{t.Fatal(err)}
  if _,ok:=report["root"];!ok {continue}
  if err=compiled.Validate(report);err!=nil {t.Fatalf("%s: %v",path,err)}
  count++
 }
 if count!=5 {t.Fatalf("expected five actual CLI reports; got %d",count)}
}
